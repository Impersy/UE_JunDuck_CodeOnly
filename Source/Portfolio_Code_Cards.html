// 입력은 바로 실행하지 않고 ActionRequest로 변환한다.
void AJunPlayer::RequestAction(EPlayerAction NextAction)
{
    FActionRequest Request;
    Request.From = CurrentAction;
    Request.To = NextAction;

    // 현재 상태에서 바로 시작할 수 있으면 즉시 실행한다.
    if (CanStartImmediately(Request))
    {
        ProcessAction(Request);
        return;
    }

    // 후딜 캔슬 타이밍이 아직 열리지 않았다면 입력을 저장한다.
    if (CanBuffer(Request))
    {
        PendingAction = Request;
    }
}

void AJunPlayer::TryProcessPendingAction()
{
    if (!PendingAction.IsSet())
        return;

    // FromAction, ToAction에 맞는 캔슬 규칙을 찾는다.
    FCancelRule Rule = FindCancelRule(PendingAction);

    // OpenTime 이전이면 아직 캔슬 불가.
    if (ElapsedTime < Rule.OpenTime)
        return;

    // 기존 몽타주를 부드럽게 빼고 다음 액션을 실행한다.
    StopCurrentMontage(Rule.BlendOutTime);
    ProcessAction(PendingAction);
    PendingAction.Reset();
}

void UDefenseComponent::HandleDefenseStarted()
{
    if (Player->IsParrySuccess())
    {
        // 패링 성공 중 재입력: 탭은 체인 패리, 홀드는 가드 전환.
        RefreshChainParryWindow();
        TryConvertToGuardHold();
        return;
    }

    if (Player->IsHitReacting())
    {
        // 피격 중에도 정식 캔슬 타이밍이면 방어로 전환한다.
        Player->TryCancelHitReactIntoDefense();
        return;
    }

    // 일반 입력은 GuardStart를 재생하며 패링 윈도우를 연다.
    StartGuardStart();
    OpenParryWindow();
}

EDefenseResult UDefenseComponent::ResolveDefense(const FAttackData& Attack)
{
    // 공격자를 바라보는 각도 안에 있어야 방어 판정 가능.
    if (!IsFacingAttacker(Attack.Attacker))
        return EDefenseResult::Hit;

    // 완벽 패리 윈도우가 일반 패리보다 우선한다.
    if (IsPerfectParryWindowOpen())
        return EDefenseResult::PerfectParry;

    if (IsParryWindowOpen())
        return EDefenseResult::NormalParry;

    if (CanGuard(Attack))
        return EDefenseResult::GuardBlock;

    return EDefenseResult::Hit;
}

void AJunBoss::RequestCombatState(EBossState NextState)
{
    // 모든 전투 상태 전환은 하나의 진입점에서 검증한다.
    if (!CanTransition(CurrentState, NextState))
        return;

    ExitState(CurrentState);
    CurrentState = NextState;
    EnterState(CurrentState);
}

FBossAttack AJunBoss::SelectNextAttack()
{
    TArray<FBossAttack> Candidates = GetAllAttacks();

    // 거리, 페이즈, 쿨타임으로 사용 가능한 공격만 남긴다.
    Candidates = FilterByDistance(Candidates, TargetDistance);
    Candidates = FilterByPhase(Candidates, CurrentPhase);
    Candidates = FilterByCooldown(Candidates);

    // 후보가 없다면 공격 대신 접근 행동으로 전환한다.
    if (Candidates.IsEmpty())
        return MakeApproachAction();

    return WeightedRandom(Candidates);
}

void AJunBoss::OnAttackLinkNotify(const FAttackLinkData& LinkData)
{
    // 몽타주 Notify 시점에서 다음 공격 링크 여부를 결정한다.
    if (!ChancePassed(LinkData.TriggerChance))
        return;

    FBossAttack NextAttack = SelectLinkCandidate(LinkData);

    if (!IsInRange(NextAttack))
    {
        // 사거리 밖이면 먼저 거리를 맞춘 뒤 공격한다.
        MoveToRangeThenAttack(NextAttack);
        return;
    }

    BlendToAttackMontage(NextAttack);
}

void UAttackTraceNotify::OnHit(AActor* Target)
{
    // 공격자는 대상 클래스를 모르고 공통 HitRequest만 만든다.
    FAttackHitRequest Request;
    Request.Attacker = Owner;
    Request.Damage = Damage;
    Request.PostureDamage = PostureDamage;
    Request.HitReactType = HitReactType;
    Request.DefenseRule = DefenseRule;

    if (IAttackTarget* HitTarget = Cast<IAttackTarget>(Target))
    {
        // 피격자는 자신의 상태에 맞게 패링, 가드, 피격을 해석한다.
        HitTarget->ReceiveAttack(Request);
    }
}

void AJunMonster::ReceiveAttack(const FAttackHitRequest& Request)
{
    // 방어 계열 결과가 먼저 처리되면 데미지/피격 리액션은 생략된다.
    if (TryParry(Request))
        return;

    if (TryGuard(Request))
        return;

    ApplyDamage(Request.Damage);

    if (HasSuperArmor())
    {
        // 슈퍼아머 중에는 몽타주 전환 대신 피지컬 히트만 적용한다.
        ApplyPhysicalHit(Request);
        return;
    }

    PlayHitReactMontage(Request.HitReactType);
    ApplyPhysicalHit(Request);
}

bool UExecutionComponent::TryStartExecution()
{
    // 처형 가능 대상을 찾고 인터페이스로 검증한다.
    IExecutionTarget* Target = FindExecutionTarget();

    if (!Target || !Target->CanBeExecutedBy(Player))
        return false;

    // 플레이어의 현재 액션을 끊고 처형 액션으로 전환한다.
    CancelCurrentPlayerAction();
    PlayPlayerExecutionMontage(Target);

    // 대상은 자신의 클래스에 맞는 처형 당하는 흐름을 실행한다.
    Target->BeginExecutedBy(Player);
    return true;
}

void AJunBoss::BeginExecutedBy(AJunPlayer* Executor)
{
    // 보스는 처형 몽타주, Life 감소, 페이즈 전환을 처리한다.
    StopAllAttackTraces();
    PlayVictimExecutionMontage(Executor);
    ConsumeBossLife();
    UpdateBossHUD();
}

void AJunTutorialNPC::BeginExecutedBy(AJunPlayer* Executor)
{
    // 튜토리얼 NPC는 죽지 않고 다음 학습 단계로 복구된다.
    PlayVictimExecutionMontage(Executor);
    RestoreHealthAndPosture();
    CompleteTutorialExecutionTask();
}
